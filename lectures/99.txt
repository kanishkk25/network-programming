1. send()

Purpose: Sends data from your program through a connected socket to the other side.

ssize_t send(
    int sockfd,
    const void *buf,
    size_t len,
    int flags
);

Parameters:

sockfd → socket through which you want to send data.
buf → pointer to the data you want to send.
len → number of bytes you want to send.
flags → special options controlling how the data is sent. Usually 0.

Return value:

positive number → number of bytes actually sent.
-1 → error.

recv()

Purpose: Receives data from the other side through a socket into your program.

ssize_t recv(
    int sockfd,
    void *buf,
    size_t len,
    int flags
);

Parameters:

sockfd → socket from which you want to receive data.
buf → memory where the received data will be stored.
len → maximum number of bytes you want to receive.
flags → special options controlling how data is received. Usually 0.

Return value:

positive number → number of bytes actually received.
0 → the other side has closed the connection.
-1 → error.

#include<stdio.h>
#include<string.h>
#include<errno.h>
#include<sys/socket.h>
#include<arpa/inet.h>
#include<netdb.h>
#include<unistd.h>

int main()
{
char request_data[1025];
char response_data[1025];

size_t request_data_size;
size_t response_data_size;

ssize_t bytes_send,bytes_received;	// s for signed

struct addrinfo hints;
int result;
struct addrinfo *ptr2top_node,*t;
int server_socket_descriptor;
struct sockaddr_storage client_socket_storage;
socklen_t client_sockaddr_size;
struct sockaddr_in *client_sockaddr_in;
int client_socket_descriptor;

char client_ip[16];
unsigned short int client_port;

// setup hints structures
memset(&hints,0,sizeof(hints));
hints.ai_family=AF_INET;
hints.ai_socktype=SOCK_STREAM;
hints.ai_flags=AI_PASSIVE;

// call to getaddrinfo
result=getaddrinfo(NULL,"9696",&hints,&ptr2top_node);
if(result==-1)
{
printf("Error : %s\n",gai_strerror(errno));
return 0;
}

for(t=ptr2top_node;t!=NULL;t=t->ai_next)
{
server_socket_descriptor=socket(t->ai_family,t->ai_socktype,t->ai_protocol);
if(server_socket_descriptor==-1) continue;

result=bind(server_socket_descriptor,t->ai_addr,t->addrlen);
if(result==-1)
{
close(server_socket_descriptor);
continue;
}
break; 	// socket creation and bind call done with no errors
}

// release linked list
freeaddrinfo(ptr2top_node);

if(t==NULL)
{
printf("Unable to create socket\n");
return 0;
}

result=listen(server_socket_descriptor,100);

{
printf("Unable to listen for connections\n");
printf("Error : %s\n",strerror(errno));

while(1)
{
printf("Server is ready to accept connection on port 9696\n");

client_sockaddr_size=sizeof(struct sockaddr_storage);

// accepting connection from backlog queue, if backlog queue is empty wait for connection

client_socket_descriptor=accept(server_socket_descriptor,(struct sockaddr *)&client_socket_storage,&client_sockaddr_size);

{
printf("Unable to accept connection request\n");
printf("Error : %s\n",strerror(errno));
continue;

printf("Connection request arrived and accepted\n");

if(client_sockaddr_size<=sizeof(struct sockaddr_storage))
{
if(client_sockaddr_storage.ss_family==AF_INET)

else
{

if(errno==EINTR)

send() - Your program → other side
recv() - Other side → your program

send() is used when your program wants to send data through a socket to the other side.
recv() is used when your program wants to receive data from the other side through a socket.
