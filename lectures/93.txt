#include<stdio.h>
#include<stdlib.h>
#include<string.h>

#define st_type unsigned int
#define ST_INDIAN 1
#define ST_AMERICAN 2
#define ST_OTHER 3

struct student
{
st_type student_type;
};

struct indian_student
{
st_type student_type;
char aadhar_number[17];	// 1 extra for \0
};
struct american_student
{
st_type student_type;
char social_security_numher[10];
};
struct other_student
{
st_type student_type;
char unique_identification_number[101];
char country_name[51];
};

struct student *queue[5];
int front;

void populate_queue()
{
struct indian_student *i_student;
struct american_student *a_student;
struct other_student *o_student;

// data of first caller
i_student=(struct indian_student *)malloc(sizeof(struct indian_student));
i_student->student_type=ST_INDIAN;
strcpy(i_student->aadhar_number,"1111222233334444");
queue[0]=(struct student *)i_student;

// data of second caller
o_student=(struct other_student *)malloc(sizeof(struct other_student));
o_student->student_type=ST_OTHER;
strcpy(o_student->unqique_identification_number,"A13KD335M");
strcpy(o_student->country_name,"Australia");
queue[1]=(struct student *)o_student;

// data of third caller
a_student=(struct american_student *)malloc(sizeof(struct american_student));
a_student->student_type=ST_AMERICAN;
strcpy(a_student->social_security_number,"123456789");
queue[2]=(struct student *)a_student;

// data of fourth caller
i_student=(struct indian_student *)malloc(sizeof(struct indian_student));
i_student->student_type=ST_INDIAN;
strcpy(i_student->aadhar_number,"8888999955551111");
queue[3]=(struct student *)i_student;

// data of fifth caller
o_student=(struct other_student *)malloc(sizeof(struct other_student));
o_student->student_type=ST_OTHER;
strcpy(o_student->unqique_identification_number,"123412348888");
strcpy(o_student->country_name,"South Africa");
queue[4]=(struct student *)o_student;

front=0;	// first fetch should serve record at 0th index
}

void release_queue()
{
int i;
for(i=0;i<=4;i++) free(queue[i]);
}

void get_next_caller_information(struct student *s,int *ptr2block_with_reservation_size)
{
struct student *student_from_queue;
int size_of_record_to_serve;
int size_of_reservation_done_by_user;

student_from_queue=queue[front];
size_of_reservation_done_by_user=*ptr2block_with_reservation_size;

if(student_from_queue->student_type==ST_INDIAN)
{
size_of_record_to_serve=sizeof(struct indian_student);
}
if(student_from_queue->student_type==ST_AMERICAN)
{
size_of_record_to_serve=sizeof(struct american_student);
}
if(student_from_queue->student_type==ST_OTHER)
{
size_of_record_to_serve=sizeof(struct other_student);
}

if(size_of_record_to_serve==size_of_reservation_done_by_user)
{
memcpy(s,student_from_queue,size_of_record_to_server);
}
if(size_of_record_to_serve<size_of_reservation_done_by_user)
{
memcpy(s,student_from_queue,size_of_record_to_server);
*ptr2block_with_reservation_size=size_of_record_to_serve;
}
if(size_of_record_to_serve>size_of_reservation_done_by_user)
{
memcpy(s,student_from_queue,size_of_reservation_done_by_user);
*ptr2block_with_reservation_size=size_of_record_to_serve;
}


front++;
}

int main()
{
struct other_student record; // we go for fattest data type

struct indian_student *indian;
struct american_student *american;
int i;
int original_size;
int updated_size
st_type student_type;
populate_queue(); // to populate dummy data of callers

for(i=1;i<=5;i++)
{

original_size=sizeof(record);
updated_size=original_size;

// calling get_next_caller_information to fetch data from queue

get_next_caller_information((struct student *)&record,&updated_size);
if(updated_size<=original_size)
{
student_type=record.student_stype;
if(student_type==ST_INDIAN)
{
indian=(struct indian_student *)&record;
printf("Indian student calling\n");
printf("Aadhar number : %s\n",indian->aadhar_number);
}
if(student_type==ST_AMERICAN)
{
indian=(struct american_student *)&record;
printf("American student calling\n");
printf("Social security number : %s\n",american->social_security_number);
}
if(student_type==ST_OTHER)
{
printf("Student is neither American or Indian\n");
printf("Srudent is from : %s\n",record.country_name);
printf("Unique identification number : %s\n",record.unique_identification_number);
}
}
if(updated_size>original_size)
{
printf("Information is incomplete\n");
}

// processing the fetched data

}



release_queue(); // to release memory
return 0;
}
