#include<stdio.h>
#include<netdb.h>
#include<sys/socket.h>
#include<arpa/inet.h>
#include<unistd.h>
#include<errno.h>
#include<string.h>

int main()
{
int result;

struct addrinfo hints;
struct addrinfo *ptr2top_node;
struct addrinfo *t;
int socket_descriptor;

memset(&hints,0,sizeof(hints));	      // very important to intialize all member to 0

hints.ai_family=AF_INET;
hints.ai_socktype=SOCK_STREAM;
hints.ai_flag=AI_PASSIVE;

// step 1 completed: we have created a structure and specified our preferences 

result=getaddrinfo(NULL,"9696",&hints,&ptr2top_node);

// getaddrinfo 0 if it succeeds

/*
1st argument is NULL and hints.ai_flag is set to AI_PASSIVE
hence whenever the socket will be bound, it will be bound to a wildcard IP which means that whenever a request arrives on any IP of this machine and the port number in request is 9696, in that case the socket will receive the request

2nd argument is port number in string form

3rd argument is the address of struct addrinfo in which we have specified our preferences

4th argument is address of a pointer of type struct adddrinfo *
	
the getaddrinfo function will generate a linked list and will assign then address of the top node to the pointer whose address we have passed as 4th argument

members of struct addrinfo are as follows -

ai_family
ai_socktype
ai_flag

ai_protocol
ai_addr
ai_addrlen
ai_cannoname
and the pointer ai_next to point the next node

we populated the first 3 members

in the linked list that getaddrinfo will create the remaining members (as required) will get populated
 
*/
for(t=ptr2top_node;t!=NULL;t=t->ai_next)
{
socket_descriptor=socket(t->ai_family,t->socktype,t->ai_protocol);
if(server_socket_descriptor==-1)
{
continue;
// socket creation failed, the information in the current node could not be implemented
// to create a socket
}
result=bind(server_socket_descriptor,t->ai_addr,t->ai_addrlen);
if(result==-1)
{
close(server_socket_descriptor);
// socket got create but bind failed, hence resource associated with created socket 
// need to be released
}
else
{
break;
}
}

freeaddrinfo(ptr2top_node);		// important to avoid memory leaks
// to free the whole linked list

if(t==NULL)	// None of the nodes in the linked list helped us in creating or binding 
{		
printf("Socket support is not available\n");
return 0;
}

// if the control reaches here, it means a socket got created 
// it it capable of accepting requests
// it has been bound to port 9696
// Now lets put the socket in listening mode


listen(server_socket_descriptor,100);
/*
1st argument : the descriptor of the created socket
2nd arguement : size of the backlog queue (we have specified 100)

we need to accept the request from underlying queue
Let say the request arrived, we accept it and we are processing it at that time 110 request arrive, they will get added to queue (only 100), 10 requests will be ignored
Note : not refused
if the request is ignore, client will retry for certain number of times
if refursed then the client won't try again. Hence it is important to understand that after the backlog queue is full and a request arrives, it will be ignored not refused
*/

return 0;
}
