technique 1 -

yeh ek simple trick hain, isme yeh nahi kahenge ke woh function return karega -

#include<stdlib.h>
#include<pthread.h>

struct data_wrapper
{
int num1;
int num2;
int sum;
};
void * sam(void *ptr)
{
struct data_wrapper *p;
p=(struct data_wrapper *)ptr;
p->sum=p->num1+p->num2;
return NULL;
}

int main()
{
struct data_wrapper s;
pthread_t thread_id;
int st;

s.num1=10;
s.num2=20;

st=pthread_create(&thread_it,NULL,sam,(void *)&s);
if(st!=0)
{
printf("Unable to create thread\n");
return 0;
}
pthread_join(thread_it,NULL);
printf("Sum is  %d\n",s.sum);
return 0;
}

pthread_join(thread_it);

main function/thread ko rukana padega jab tak sam thread ka kaam nahi ho jaata hain

yeh koi return karana nahi hua

{
int pthread_join(pthread_t thread_id,void **retval)

Purpose: Waits for a specific thread to finish.

pthread_join(thread_id,(void **)return_value);

thread_id → id of the thread to wait for.
return_value → Address where the thread's return value is stored. Use NULL if you don't need it.

Returns: 0 on success; a non-zero error code on failure.

pthread_join() → wait for a thread to finish and optionally collect its result.
}

technique 3 -

yeh right way hain -

#include<stdlib.h>
#include<pthread.h>
#include<stdlib.h>

struct data_wrapper
{
int num1;
int num2;
int sum;
};
void * sam(void *ptr)
{
struct data_wrapper *p;
int *k;
p=(struct data_wrapper *)ptr;

k=(int *)malloc(sizeof(int));
if(k!=NULL)
{
*k=p->num1+p->num2;
return (void *)k;
}
else
{
return NULL;
}
}

int main()
{
struct data_wrapper s;
pthread_t thread_id;
int st;
int *x;
s.num1=10;
s.num2=20;

st=pthread_create(&thread_it,NULL,sam,(void *)&s);
if(st!=0)
{
printf("Unable to create thread\n");
return 0;
}

pthread_join(thread_it,(void **)&x);
printf("Sum is  %d\n",*x);

free(x) 	// important to avoid memory leak

return 0;
}

void * sam(void *ptr)
{

{

else
{
return NULL;
}
}

{
struct data_wrapper s;

{
printf("Unable to create thread\n");
return 0;

return karane ki bajaye pthread_exit ko call kiya for exiting from thread 


pthread_exit()
Purpose: Ends the calling thread.

pthread_exit((void *)return_value);

return_value → Value that thread returns to pthread_join(). Can be NULL.
Returns: Nothing (void).

In short: pthread_exit() → stop the current thread and optionally give a result to the thread that joins it.
