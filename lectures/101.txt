ab maanlo main thread se 5 thread create karane hain aur har thread ko ek number bhejana hain

toh pehle agar main thread create hua phir yeh 5 thread mein se 1st thread create huye toh iska matalab nahi hain ki yeh 1st thread ka code pehle chal padega

jab bhi naya thread create kiya jaayega toh internal queue re-shuffle hogi


#include<stdio.h>
#include<pthread.h>
void * sam(void *ptr)
{
int *p;
p=(int *)ptr;
printf("%d\n",*p);
return NULL;
}
int main()
{
pthread_t thread_id[5];
int a,b;
b=101;
for(a=0;a<=4;a++)
{
pthread_create(&thread_id[a],NULL,sam,&b);
b++;
}
for(a=0;<=4;a++)
{
pthread_join(thread_id[a],NULL);
}
return 0;
}

hum in threads ke paas number nahi pahucha rahe hain, address pahucha rahe hain b ka, toh jab kisi thread ka number aayega tab b ki jo value hogi woh print hogi

#include<stdio.h>
#include<pthread.h>
void * sam(void *ptr)
{
int *p;
p=(int *)ptr;
printf("%d\n",*p);
return NULL;
}
int main()
{
pthread_t thread_id[5];
int numbers[5];
int a,b;
b=101;
for(a=0;a<=4;a++)
{
numbers[a]=b;
pthread_create(&thread_id[a],NULL,sam,&(void *)numbers[a]);
b++;
}
for(a=0;<=4;a++)
{
pthread_join(thread_id[a],NULL);
}
return 0;
}
